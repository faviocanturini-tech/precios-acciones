import pandas as pd
import numpy as np

# Cargar datos
df = pd.read_csv('data/auto_update_log.csv')

# RSI calculation
def calcular_rsi(df_ticker, periodo=14):
    delta = df_ticker['Close'].diff()
    gain = (delta.where(delta > 0, 0)).rolling(window=periodo).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(window=periodo).mean()
    rs = gain / loss
    rsi = 100 - (100 / (1 + rs))
    return rsi.iloc[-1]

# Tendencia calculation
def calcular_tendencia(df_ticker, dias):
    if len(df_ticker) < dias:
        return 0
    cierre_actual = df_ticker['Close'].iloc[-1]
    cierre_anterior = df_ticker['Close'].iloc[-dias]
    return ((cierre_actual - cierre_anterior) / cierre_anterior) * 100

# Para cada ticker de IBKR-UK Real
tickers_ibkr_real = ['AMZN', 'GOOGL', 'PLTR', 'AVGO', 'NVDA', 'OXY', 'AAPL']

# Cargar datos
df = pd.read_csv('data/auto_update_log.csv')

# RSI calculation
def calcular_rsi(df_ticker, periodo=14):
    delta = df_ticker['Close'].diff()
    gain = (delta.where(delta > 0, 0)).rolling(window=periodo).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(window=periodo).mean()
    rs = gain / loss
    rsi = 100 - (100 / (1 + rs))
    return rsi.iloc[-1]

# Tendencia calculation
def calcular_tendencia(df_ticker, dias):
    if len(df_ticker) < dias:
        return 0
    cierre_actual = df_ticker['Close'].iloc[-1]
    cierre_anterior = df_ticker['Close'].iloc[-dias]
    return ((cierre_actual - cierre_anterior) / cierre_anterior) * 100

import pandas as pd
import json

# Tickers de interés (todos los de IBKR y TYBA)

# Últimos 30 días para cada ticker
result = {}
for ticker in tickers:
    ticker_data = df[df['Ticker'] == ticker].tail(30)
    if len(ticker_data) > 0:
        closes = ticker_data['Close'].values
        var_pcts = ticker_data['% var.'].values
        
        # Calcular RSI (14 períodos)
        deltas = pd.Series(closes).diff()
        gains = deltas.where(deltas > 0, 0)
        losses = (-deltas.where(deltas < 0, 0))
        avg_gain = gains.rolling(14).mean().iloc[-1]
        avg_loss = losses.rolling(14).mean().iloc[-1]
        if avg_loss != 0:
            rs = avg_gain / avg_loss
            rsi = 100 - (100 / (1 + rs))
        else:
            rsi = 100
        
        # Tendencia 10d y 30d
        if len(closes) >= 10:
            tend_10d = ((closes[-1] / closes[-10]) - 1) * 100
        else:
            tend_10d = 0
        if len(closes) >= 30:
            tend_30d = ((closes[-1] / closes[0]) - 1) * 100
        else:
            tend_30d = ((closes[-1] / closes[0]) - 1) * 100
            
        # Variación 5d
        if len(closes) >= 5:
            var_5d = ((closes[-1] / closes[-5]) - 1) * 100
        else:
            var_5d = 0
        
        result[ticker] = {
            'cierre': round(closes[-1], 2),
            'rsi': round(rsi, 1),
            'tend_10d': round(tend_10d, 1),
            'tend_30d': round(tend_30d, 1),
            'var_5d': round(var_5d, 1),
            'max_30d': round(max(closes), 2),
            'min_30d': round(min(closes), 2)
        }

print(json.dumps(result, indent=2))
